function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // Jika ini pengiriman data satuan (transaksi baru) if (!Array.isArray(data)) { sheet.appendRow([ data.noTrx || "-", new Date(), data.customerName || "-", data.customerPhone || "-", data.totalAmount || 0, data.paymentStatus || "LUNAS", data.cashier || "-" ]); } // Jika ini pengiriman data banyak sekaligus (migrasi data lama) else { data.forEach(function(item) { sheet.appendRow([ item.noTrx || "-", item.createdAt ? new Date(item.createdAt) : new Date(), item.customerName || "-", item.customerPhone || "-", item.totalAmount || 0, item.paymentStatus || "LUNAS", item.cashier || "-" ]); }); } return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); }
